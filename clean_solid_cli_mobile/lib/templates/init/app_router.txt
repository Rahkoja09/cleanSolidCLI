import 'package:flutter/material.dart';

import 'package:go_router/go_router.dart';

/// Tracker d'initialisation — définit ici pour éviter les imports circulaires.
final appInitReadyProvider = StateProvider<bool>((ref) => false);

/// GoRouter en tant que Riverpod Provider.
/// Cela permet de reagir aux changements d'etat (auth, etc.)
/// et de declencher des redirections automatiquement.
///
/// Quand vous ajoutez auth avec "cscm auth", les routes et
/// la logique de redirect seront ajoutees ici.
final goRouterProvider = Provider<GoRouter>((ref) {
  final isReady = ref.watch(appInitReadyProvider);

  return GoRouter(
    initialLocation: '/',
    debugLogDiagnostics: true,
    refreshListenable: _GoRouterRefresh(ref),
    routes: [
      GoRoute(
        path: '/',
        builder: (context, state) => const _PlaceholderPage(),
      ),
      // [ROUTES_ANCHOR]
    ],
    redirect: (context, state) {
      // Attendre que l'app soit initialisee
      if (!isReady) return '/';
      return null;
    },
    errorBuilder: (context, state) => Scaffold(
      body: Center(
        child: Text('Page non trouvée: ${state.uri.path}'),
      ),
    ),
  );
});
