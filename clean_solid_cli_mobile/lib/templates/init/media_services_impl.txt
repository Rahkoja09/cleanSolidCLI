import 'dart:io';

import 'package:flutter/material.dart';
import 'package:flutter_image_compress/flutter_image_compress.dart';
import 'package:image_picker/image_picker.dart';
import 'package:mime/mime.dart';
import 'package:path/path.dart' as p;
import 'package:path_provider/path_provider.dart';
import 'package:supabase_flutter/supabase_flutter.dart';
import 'package:video_compress/video_compress.dart';
import 'package:video_thumbnail/video_thumbnail.dart';

import 'package:{{name}}/core/error/exceptions.dart';
import 'package:{{name}}/core/network/network_info.dart';
import 'package:{{name}}/core/utils/is_url.dart';
import 'package:{{name}}/core/services/media_services.dart';

class MediaServiceImpl implements MediaServices {
  final SupabaseClient _supabase;
  final NetworkInfo _networkInfo;
  final ImagePicker _picker;

  MediaServiceImpl(this._supabase, this._networkInfo, this._picker);

  // ═══════════════════════════════════════
  // PICKING
  // ═══════════════════════════════════════

  @override
  Future<File?> pickImage({required bool fromGallery}) async {
    final XFile? picked = await _picker.pickImage(
      source: fromGallery ? ImageSource.gallery : ImageSource.camera,
    );
    return picked != null ? File(picked.path) : null;
  }

  @override
  Future<List<File>> pickMultiImages() async {
    final files = await _picker.pickMultiImage();
    return files.map((f) => File(f.path)).toList();
  }

  @override
  Future<File?> pickVideo({required bool fromGallery}) async {
    final XFile? picked = await _picker.pickVideo(
      source: fromGallery ? ImageSource.gallery : ImageSource.camera,
    );
    return picked != null ? File(picked.path) : null;
  }

  // ═══════════════════════════════════════
  // PROCESSING
  // ═══════════════════════════════════════

  @override
  Future<File> compressImage(File file, {int quality = 80}) async {
    try {
      final targetPath = p.join(
        file.parent.path,
        '${p.basenameWithoutExtension(file.path)}_comp.webp',
      );
      final result = await FlutterImageCompress.compressAndGetFile(
        file.absolute.path,
        targetPath,
        quality: quality,
        format: CompressFormat.webp,
      );
      return result != null ? File(result.path) : file;
    } catch (e) {
      debugPrint('Image compression failed: $e');
      return file;
    }
  }

  @override
  Future<File?> compressVideo(File file) async {
    try {
      final info = await VideoCompress.compressVideo(
        file.path,
        quality: VideoQuality.MediumQuality,
        includeAudio: true,
      );
      return info?.file ?? file;
    } catch (e) {
      debugPrint('Video compression failed: $e');
      return file;
    }
  }

  @override
  Future<File?> generateVideoThumbnail(dynamic videoFile) async {
    try {
      final isRemote = isUrl(videoFile as String);
      final videoPath = isRemote
          ? videoFile as String
          : (videoFile as File).path;

      final bytes = await VideoThumbnail.thumbnailData(
        video: videoPath,
        imageFormat: ImageFormat.JPEG,
        maxWidth: 200,
        quality: 5,
      );
      if (bytes == null) return null;

      final tempDir = await getTemporaryDirectory();
      final fileName = isRemote
          ? Uri.parse(videoPath).pathSegments.last
          : videoPath.split('/').last;
      final thumbFile = File('${tempDir.path}/$fileName.jpg');
      await thumbFile.writeAsBytes(bytes);
      return thumbFile;
    } catch (e) {
      debugPrint('Thumbnail generation failed: $e');
      return null;
    }
  }

  @override
  void validateMediaSize(
    File file, {
    int maxImageSizeMB = 5,
    int maxVideoSizeMB = 50,
  }) {
    final mimeType = lookupMimeType(file.path);
    final isVideo = mimeType?.startsWith('video/') ?? false;
    final maxSize = isVideo ? maxVideoSizeMB : maxImageSizeMB;

    if (file.lengthSync() > maxSize * 1024 * 1024) {
      throw ApiException(
        message: 'File too large. Max: ${isVideo ? maxVideoSizeMB : maxImageSizeMB}MB',
      );
    }
  }

  // ═══════════════════════════════════════
  // STORAGE (SUPABASE)
  // ═══════════════════════════════════════

  @override
  Future<String> uploadFile({
    required File file,
    required String bucket,
    required String path,
  }) async {
    if (!(await _networkInfo.isConnected)) {
      throw NetworkException(message: 'No internet connection');
    }

    final contentType = lookupMimeType(file.path);
    try {
      await _supabase.storage.from(bucket).upload(
        path,
        file,
        fileOptions: FileOptions(contentType: contentType, upsert: true),
      );
      return getPublicUrl(path, bucket);
    } on StorageException catch (e) {
      throw StorageExceptions(message: e.message, code: e.statusCode);
    } on ServerException catch (e) {
      throw ServerException(message: e.message, code: e.code);
    } catch (e) {
      throw UnexceptedException(message: '$e');
    }
  }

  @override
  String getPublicUrl(String filePath, String bucketName) {
    try {
      return _supabase.storage.from(bucketName).getPublicUrl(filePath);
    } on StorageException catch (e) {
      throw StorageExceptions(message: e.message, code: e.statusCode);
    } catch (e) {
      throw UnexceptedException(message: '$e');
    }
  }

  @override
  Future<void> deleteFile(String filePath, String bucketName) async {
    try {
      await _supabase.storage.from(bucketName).remove([filePath]);
    } on StorageException catch (e) {
      throw StorageExceptions(message: e.message, code: e.statusCode);
    } catch (e) {
      throw UnexceptedException(message: '$e');
    }
  }

  // ═══════════════════════════════════════
  // CLEANUP
  // ═══════════════════════════════════════

  @override
  Future<void> clearCache() async {
    await VideoCompress.deleteAllCache();
  }
}
