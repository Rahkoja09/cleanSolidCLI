/// Evaluates password strength and returns a [PasswordStrength] enum.
///
///   Weak    : < 6 chars
///   Medium  : >= 6 chars, some complexity
///   Strong  : >= 10 chars, uppercase + digits + special chars

enum PasswordStrength { weak, medium, strong }

PasswordStrength evaluatePasswordStrength(String password) {
  if (password.length < 6) return PasswordStrength.weak;

  final hasUppercase = password.contains(RegExp(r'[A-Z]'));
  final hasLowercase = password.contains(RegExp(r'[a-z]'));
  final hasDigits = password.contains(RegExp(r'\d'));
  final hasSpecial = password.contains(RegExp(r'[!@#\$%^&*(),.?":{}|<>]'));

  final score = [hasUppercase, hasLowercase, hasDigits, hasSpecial]
      .where((b) => b).length;

  if (password.length >= 10 && score >= 3) {
    return PasswordStrength.strong;
  } else if (score >= 2) {
    return PasswordStrength.medium;
  }
  return PasswordStrength.weak;
}
